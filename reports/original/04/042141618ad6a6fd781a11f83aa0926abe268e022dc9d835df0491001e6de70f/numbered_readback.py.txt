 160 def number(bits):
 161  if len(bits)!=16:raise ValueError('SCORE_BITS')
 162  v=struct.unpack('<d',bytes.fromhex(bits))[0]
 163  if not math.isfinite(v):raise ValueError('NONFINITE_SCORE')
 164  return v
 165 def canonical(text):
 166  d={}

 170 def verify_proposal(alphabet,current,cw,match,claimed):
 171  # Independent mapping witness for the unchanged sequence of transpositions/replacements.
 172  mapping=dict(zip(alphabet,current));owners={v:k for k,v in mapping.items()}
 173  for symbol,letter in zip(cw,match):
 174   if symbol.isspace() or letter.isspace():continue
 175   old=mapping[symbol]
 176   if old==letter:continue
 177   other=owners.get(letter)
 178   if other is not None:mapping[other]=old;owners[old]=other
 179   else:del owners[old]
 180   mapping[symbol]=letter;owners[letter]=symbol
 181  if len(cw)!=len(match) or ''.join(mapping[c] for c in alphabet)!=claimed:raise ValueError('ASSUME_WITNESS')
